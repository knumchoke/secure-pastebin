# WS4 — Jigsaw Challenge Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the server-side jigsaw/slider human-verification challenge: procedural PNG generation in pure Go, a Redis-backed single-use challenge and token store, and the `challenge.Challenger` service with server-measured solve time.

**Architecture:** `internal/challenge/jigsaw.go` renders a background with a darkened jigsaw-shaped hole and a matching piece from a seeded RNG (deterministic in tests). `internal/challenge/redis_store.go` keeps `chal:{id}` and `ctok:{token}` with TTLs, consumed via `GETDEL`. `internal/challenge/service.go` implements `Challenger` and emits audit events. HTTP handlers are WS5.

**Tech Stack:** Go stdlib `image`, `image/color`, `image/png`, `math/rand/v2` (seeded from `crypto/rand` in production), `github.com/redis/go-redis/v9`; tests: `miniredis`.

**Spec:** `docs/superpowers/specs/2026-09-11-secure-pastebin-design.md` — §9, §5.2 (`chal:`/`ctok:` keys), D8, §12 (`CHALLENGE_*`).

## Global Constraints

- Implement exactly `challenge.Challenger` from `internal/challenge/ports.go` (WS1). Compile-time assertion required.
- Images are generated procedurally at issue time (no bundled assets, no external fetches). Background 320×160 px; piece bounding box 66×66 px (50 px body + 8 px knob on each side); PNG output.
- The client never sends timing. Solve time = `now − issued_at` from the stored record; must be ≥ `CHALLENGE_MIN_SOLVE_MS`.
- Challenge and token are single-use (`GETDEL`), bound to the issuing user, TTL `CHALLENGE_TTL_SECONDS`.
- Tolerance check: `|x − stored_x| ≤ CHALLENGE_TOLERANCE_PX`.
- Audit events: `challenge_issued`, `challenge_passed`, `challenge_failed` (details: `reason` ∈ `too_fast|wrong_position|not_found`). No image bytes or coordinates in audit.
- `math/rand/v2` is only acceptable here because it is seeded from `crypto/rand`; gosec G404 is excluded repo-wide for this reason — do not use `math/rand` anywhere else.
- Shell commands are prefixed with `rtk`.

package challenge

import (
	"bytes"
	"errors"
	"image"
	"image/color"
	"image/png"
	"math"
	"math/rand/v2"
)

var errBadParams = errors.New("challenge: params leave no room for the piece")

// Params control the puzzle geometry.
type Params struct {
	Width, Height int
	Body          int // side of the square piece body
	Knob          int // radius of the tab/notch circles
}

func DefaultParams() Params { return Params{Width: 320, Height: 160, Body: 50, Knob: 8} }

// Rendered is one generated puzzle.
type Rendered struct {
	BackgroundPNG []byte
	PiecePNG      []byte
	X, Y          int
	Box           int
}

// InPiece reports whether (px,py) in piece-box coordinates belongs to the
// piece: a Body×Body square offset by Knob, plus a circular tab on the right
// edge, minus a circular notch on the top edge.
func InPiece(px, py int, p Params) bool {
	k, b := p.Knob, p.Body
	inSquare := px >= k && px < k+b && py >= k && py < k+b
	// tab: circle centred on the right edge midpoint
	dx, dy := float64(px-(k+b)), float64(py-(k+b/2))
	inTab := math.Hypot(dx, dy) <= float64(k)
	// notch: circle centred on the top edge midpoint, removed from the square
	nx, ny := float64(px-(k+b/2)), float64(py-k)
	inNotch := math.Hypot(nx, ny) <= float64(k)
	return (inSquare && !inNotch) || inTab
}

// background draws a soft gradient with random translucent blobs and grain so
// the hole edge is not trivially detectable by a flat-colour scan.
func background(rng *rand.Rand, p Params) *image.RGBA {
	img := image.NewRGBA(image.Rect(0, 0, p.Width, p.Height))
	c0 := color.RGBA{uint8(60 + rng.IntN(120)), uint8(60 + rng.IntN(120)), uint8(60 + rng.IntN(120)), 255}
	c1 := color.RGBA{uint8(60 + rng.IntN(120)), uint8(60 + rng.IntN(120)), uint8(60 + rng.IntN(120)), 255}
	for y := 0; y < p.Height; y++ {
		for x := 0; x < p.Width; x++ {
			t := float64(x+y) / float64(p.Width+p.Height)
			img.SetRGBA(x, y, color.RGBA{
				uint8(float64(c0.R)*(1-t) + float64(c1.R)*t),
				uint8(float64(c0.G)*(1-t) + float64(c1.G)*t),
				uint8(float64(c0.B)*(1-t) + float64(c1.B)*t), 255})
		}
	}
	// blobs
	for i := 0; i < 12+rng.IntN(8); i++ {
		cx, cy, r := rng.IntN(p.Width), rng.IntN(p.Height), 10+rng.IntN(40)
		tint := color.RGBA{uint8(rng.IntN(256)), uint8(rng.IntN(256)), uint8(rng.IntN(256)), 255}
		for y := cy - r; y <= cy+r; y++ {
			for x := cx - r; x <= cx+r; x++ {
				if x < 0 || y < 0 || x >= p.Width || y >= p.Height {
					continue
				}
				if math.Hypot(float64(x-cx), float64(y-cy)) > float64(r) {
					continue
				}
				o := img.RGBAAt(x, y)
				img.SetRGBA(x, y, color.RGBA{blend(o.R, tint.R), blend(o.G, tint.G), blend(o.B, tint.B), 255})
			}
		}
	}
	// grain
	for y := 0; y < p.Height; y++ {
		for x := 0; x < p.Width; x++ {
			n := int(rng.IntN(31)) - 15
			o := img.RGBAAt(x, y)
			img.SetRGBA(x, y, color.RGBA{clamp(int(o.R) + n), clamp(int(o.G) + n), clamp(int(o.B) + n), 255})
		}
	}
	return img
}

func blend(a, b uint8) uint8 { return uint8((int(a)*7 + int(b)*3) / 10) }

func clamp(v int) uint8 {
	if v < 0 {
		return 0
	}
	if v > 255 {
		return 255
	}
	return uint8(v)
}

// Render produces a puzzle from rng (seed from crypto/rand in production).
func Render(rng *rand.Rand, p Params) (Rendered, error) {
	box := p.Body + 2*p.Knob
	minX := p.Width / 3
	maxX := p.Width - box - 4
	minY, maxY := 4, p.Height-box-4
	if maxX <= minX || maxY <= minY {
		return Rendered{}, errBadParams
	}
	x := minX + rng.IntN(maxX-minX+1)
	y := minY + rng.IntN(maxY-minY+1)

	bg := background(rng, p)
	piece := image.NewRGBA(image.Rect(0, 0, box, box))
	for py := 0; py < box; py++ {
		for px := 0; px < box; px++ {
			if !InPiece(px, py, p) {
				continue
			}
			o := bg.RGBAAt(x+px, y+py)
			piece.SetRGBA(px, py, o)
			// hole: darken to 40 % and desaturate slightly
			bg.SetRGBA(x+px, y+py, color.RGBA{uint8(int(o.R) * 2 / 5), uint8(int(o.G) * 2 / 5), uint8(int(o.B) * 2 / 5), 255})
		}
	}
	var bgBuf, pcBuf bytes.Buffer
	enc := png.Encoder{CompressionLevel: png.BestSpeed}
	if err := enc.Encode(&bgBuf, bg); err != nil {
		return Rendered{}, err
	}
	if err := enc.Encode(&pcBuf, piece); err != nil {
		return Rendered{}, err
	}
	return Rendered{BackgroundPNG: bgBuf.Bytes(), PiecePNG: pcBuf.Bytes(), X: x, Y: y, Box: box}, nil
}

// Record is what the server remembers about an issued challenge. X is the secret.
type Record struct {
	UserID   uuid.UUID `json:"user_id"`

	SaveChallenge(ctx context.Context, id string, rec Record, ttl time.Duration) error
	TakeChallenge(ctx context.Context, id string) (Record, error)

package challenge

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"time"

	"github.com/google/uuid"
	"github.com/redis/go-redis/v9"
)

// RedisStore keeps chal:{id} and ctok:{token} (spec §5.2).
type RedisStore struct{ c *redis.Client }

var _ Store = (*RedisStore)(nil)

func NewRedisStore(c *redis.Client) *RedisStore { return &RedisStore{c: c} }

func (s *RedisStore) SaveChallenge(ctx context.Context, id string, rec Record, ttl time.Duration) error {
	b, err := json.Marshal(rec)
	if err != nil {
		return err
	}
	if err := s.c.Set(ctx, "chal:"+id, b, ttl).Err(); err != nil {
		return fmt.Errorf("challenge: save: %w", err)
	}
	return nil
}

func (s *RedisStore) TakeChallenge(ctx context.Context, id string) (Record, error) {
	var rec Record
	if id == "" {
		return rec, ErrChallengeNotFound
	}
	b, err := s.c.GetDel(ctx, "chal:"+id).Bytes()
	if errors.Is(err, redis.Nil) {
		return rec, ErrChallengeNotFound
	}
	if err != nil {
		return rec, fmt.Errorf("challenge: take: %w", err)
	}
	if err := json.Unmarshal(b, &rec); err != nil {
		return rec, ErrChallengeNotFound
	}
	return rec, nil
}

func (s *RedisStore) SaveToken(ctx context.Context, token string, userID uuid.UUID, ttl time.Duration) error {
	if err := s.c.Set(ctx, "ctok:"+token, userID.String(), ttl).Err(); err != nil {
		return fmt.Errorf("challenge: save token: %w", err)
	}
	return nil
}

func (s *RedisStore) TakeToken(ctx context.Context, token string) (uuid.UUID, error) {
	if token == "" {
		return uuid.Nil, ErrTokenInvalid
	}
	v, err := s.c.GetDel(ctx, "ctok:"+token).Result()
	if errors.Is(err, redis.Nil) {
		return uuid.Nil, ErrTokenInvalid
	}
	if err != nil {
		return uuid.Nil, fmt.Errorf("challenge: take token: %w", err)
	}
	id, err := uuid.Parse(v)
	if err != nil {
		return uuid.Nil, ErrTokenInvalid
	}
	return id, nil
}

package challenge

import (
	"context"
	crand "crypto/rand"
	"encoding/binary"
	"math/rand/v2"
	"time"

	"github.com/google/uuid"

	"github.com/knumchoke/secure-pastebin/internal/audit"
	"github.com/knumchoke/secure-pastebin/internal/auth"
	"github.com/knumchoke/secure-pastebin/internal/config"
)

type Config struct {
	TolerancePx int
	TTL         time.Duration
	MinSolve    time.Duration
	Params      Params
}

func ConfigFrom(c config.ChallengeConfig) Config {
	return Config{TolerancePx: c.TolerancePx, TTL: c.TTL, MinSolve: c.MinSolve}
}

// Service implements Challenger (spec §9).
type Service struct {
	cfg   Config
	store Store
	sink  audit.Sink
	now   func() time.Time
	seed  func() (uint64, uint64)
}

var _ Challenger = (*Service)(nil)

func cryptoSeed() (uint64, uint64) {
	var b [16]byte
	if _, err := crand.Read(b[:]); err != nil {
		panic("crypto/rand failed: " + err.Error())
	}
	return binary.LittleEndian.Uint64(b[:8]), binary.LittleEndian.Uint64(b[8:])
}

func NewService(cfg Config, store Store, sink audit.Sink, now func() time.Time, seed func() (uint64, uint64)) *Service {
	if cfg.Params == (Params{}) {
		cfg.Params = DefaultParams()
	}
	if now == nil {
		now = func() time.Time { return time.Now().UTC() }
	}
	if seed == nil {
		seed = cryptoSeed
	}
	return &Service{cfg: cfg, store: store, sink: sink, now: now, seed: seed}
}

func (s *Service) Issue(ctx context.Context, userID uuid.UUID) (Issued, error) {
	a, b := s.seed()
	r, err := Render(rand.New(rand.NewPCG(a, b)), s.cfg.Params)
	if err != nil {
		return Issued{}, err
	}
	id := auth.RandomToken(16)
	now := s.now()
	if err := s.store.SaveChallenge(ctx, id, Record{UserID: userID, X: r.X, IssuedAt: now}, s.cfg.TTL); err != nil {
		return Issued{}, err
	}
	s.sink.Record(ctx, audit.Event{At: now, Event: audit.ChallengeIssued, ActorID: &userID, Outcome: audit.OutcomeSuccess})
	return Issued{ID: id, BackgroundPNG: r.BackgroundPNG, PiecePNG: r.PiecePNG, PieceY: r.Y,
		Width: s.cfg.Params.Width, Height: s.cfg.Params.Height, ExpiresIn: s.cfg.TTL}, nil
}

func (s *Service) fail(ctx context.Context, userID uuid.UUID, reason string, err error) (string, error) {
	s.sink.Record(ctx, audit.Event{At: s.now(), Event: audit.ChallengeFailed, ActorID: &userID, Outcome: audit.OutcomeFailure,
		Details: map[string]any{"reason": reason}})
	return "", err
}

func (s *Service) Verify(ctx context.Context, userID uuid.UUID, challengeID string, x int) (string, error) {
	rec, err := s.store.TakeChallenge(ctx, challengeID)
	if err != nil || rec.UserID != userID {
		return s.fail(ctx, userID, "not_found", ErrChallengeNotFound)
	}
	now := s.now()
	if now.Sub(rec.IssuedAt) < s.cfg.MinSolve {
		return s.fail(ctx, userID, "too_fast", ErrChallengeFailed)
	}
	if d := x - rec.X; d > s.cfg.TolerancePx || d < -s.cfg.TolerancePx {
		return s.fail(ctx, userID, "wrong_position", ErrChallengeFailed)
	}
	token := auth.RandomToken(32)
	if err := s.store.SaveToken(ctx, token, userID, s.cfg.TTL); err != nil {
		return "", err
	}
	s.sink.Record(ctx, audit.Event{At: now, Event: audit.ChallengePassed, ActorID: &userID, Outcome: audit.OutcomeSuccess})
	return token, nil
}

func (s *Service) Consume(ctx context.Context, userID uuid.UUID, token string) error {
	owner, err := s.store.TakeToken(ctx, token)
	if err != nil {
		return err
	}
	if owner != userID {
		return ErrTokenInvalid
	}
	return nil
}
