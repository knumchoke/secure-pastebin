# WS6 — UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the single-page UI (Vite + vanilla TypeScript, no framework) that implements spec §8 against `docs/api/openapi.yaml`: login, new paste with live byte counter and jigsaw challenge, paste view with unlock/copy/download/verify/delete, my pastes, and a standalone verify page — with Thai/English strings and a CSP-compatible build.

**Architecture:** `ui/src/api.ts` is the only module that talks HTTP (typed client, CSRF header, error envelope). Pure utilities (`bytes.ts`, `hash.ts`, `i18n.ts`) are unit-tested with vitest under Node. Pages are functions `(root: HTMLElement, ctx: AppContext) => void | Promise<void>` mounted by a tiny history router. No inline scripts or `style=""` attributes (CSP); positioning is done through the CSSOM.

**Tech Stack:** TypeScript 5, Vite 6, vitest 2 (Node environment, Web Crypto from `globalThis.crypto`). No runtime dependencies.

**Spec:** `docs/superpowers/specs/2026-09-11-secure-pastebin-design.md` — §8, §9 (client half), §7 (contract), D9, acceptance criteria 3, 7.

## Global Constraints

- Contract: field names from `docs/api/openapi.yaml` (WS1). Base path `/api/v1`. All non-GET calls send `X-CSRF-Token` from `/auth/me`.
- `content` from the API **includes** the leading U+FEFF; strip it for display; use the canonical string for hashing and download.
- Byte counter = `3 + utf8ByteLength(textWithoutBOM)`; if the textarea already begins with U+FEFF do not add 3 again. Red + disabled Save when `> paste_max_size_bytes`. Debounce 50 ms.
- Verify page canonicalises identically (prepend BOM if absent, nothing else) before hashing.
- No inline `<script>`; no `style=""` attributes in markup; no external CDN assets; no analytics.
- i18n: `en` (default) and `th`; toggle persisted in `localStorage["pb.lang"]`, wrapped in try/catch.
- Password inputs use `autocomplete="off"` (paste password) and `autocomplete="current-password"` (login).
- Shell commands are prefixed with `rtk`.

```ts
export const BOM = "﻿";
const encoder = new TextEncoder();

export function stripBOM(s: string): string {
  return s.startsWith(BOM) ? s.slice(1) : s;
}

export function withBOM(s: string): string {
  return s.startsWith(BOM) ? s : BOM + s;
}

export function utf8Length(s: string): number {
  return encoder.encode(s).length;
}

export function canonicalByteLength(s: string): number {
  return utf8Length(withBOM(s));
}

export function canonicalBytes(s: string): Uint8Array {
  return encoder.encode(withBOM(s));
}

function trim(n: number): string {
  const s = n.toFixed(1);
  return s.endsWith(".0") ? s.slice(0, -2) : s;
}

export function formatSize(n: number): string {
  if (n >= 1 << 20) return `${trim(n / (1 << 20))} MB`;
  if (n >= 1 << 10) return `${trim(n / (1 << 10))} KB`;
  return `${n} B`;
}
```

```ts
export async function sha256Hex(bytes: Uint8Array): Promise<string> {
  const digest = await crypto.subtle.digest("SHA-256", bytes);
  return Array.from(new Uint8Array(digest), (b) => b.toString(16).padStart(2, "0")).join("");
}
```

```ts
    this.retryAfter = retryAfter;

const BASE = "/api/v1";

export class ApiClient {
  csrf: string | null = null;

  private async call<T>(method: string, path: string, body?: unknown, accept = "application/json"): Promise<T> {
    const headers: Record<string, string> = { Accept: accept };
    if (body !== undefined) headers["Content-Type"] = "application/json";
    if (method !== "GET" && this.csrf) headers["X-CSRF-Token"] = this.csrf;
    const res = await fetch(BASE + path, {
      method,
      headers,
      credentials: "same-origin",
      body: body === undefined ? undefined : JSON.stringify(body),
    });
    if (res.status === 204) return undefined as T;
    if (!res.ok) {
      let code = "http_error";
      let message = `HTTP ${res.status}`;
      let details: Record<string, unknown> = {};
      try {
        const e = (await res.json()) as { error?: { code: string; message: string; details?: Record<string, unknown> } };
        if (e.error) ({ code, message } = e.error), (details = e.error.details ?? {});
      } catch {
        /* non-JSON error body */
      }
      const ra = res.headers.get("retry-after");
      throw new ApiError(res.status, code, message, details, ra ? Number(ra) : undefined);
    }
    return (await res.json()) as T;
  }

  getConfig() {
    return this.call<PublicConfig>("GET", "/config");
  }
  login(username: string, password: string) {
    return this.call<{ user: User; csrf_token: string }>("POST", "/auth/login", { username, password });
  }
  logout() {
    return this.call<void>("POST", "/auth/logout");
  }
  me() {
    return this.call<{ user: User; csrf_token: string }>("GET", "/auth/me");
  }
  oidcStartUrl() {
    return BASE + "/auth/oidc/start";
  }
  issueChallenge() {
    return this.call<ChallengeIssued>("POST", "/challenges");
  }
  verifyChallenge(id: string, x: number) {
    return this.call<{ challenge_token: string; expires_in: number }>("POST", `/challenges/${encodeURIComponent(id)}/verify`, { x });
  }
  createPaste(body: { content: string; password?: string; ttl_seconds?: number; challenge_token?: string }) {
    return this.call<PasteMeta>("POST", "/pastes", body);
  }
  getPaste(id: string) {
    return this.call<PasteMeta>("GET", `/pastes/${encodeURIComponent(id)}`);
  }
  unlockPaste(id: string, password: string) {
    return this.call<PasteMeta>("POST", `/pastes/${encodeURIComponent(id)}/unlock`, { password });
  }
  verifyPaste(id: string, sha256: string) {
    return this.call<VerifyResult>("POST", `/pastes/${encodeURIComponent(id)}/verify`, { sha256 });

  }
```

```ts
export function h<K extends keyof HTMLElementTagNameMap>(
  tag: K,
  attrs: Record<string, string | boolean | EventListener> = {},
  ...children: (Node | string | null | undefined)[]
): HTMLElementTagNameMap[K] {
  const el = document.createElement(tag);
  for (const [k, v] of Object.entries(attrs)) {
    if (typeof v === "function") el.addEventListener(k.replace(/^on/, "").toLowerCase(), v);
    else if (typeof v === "boolean") {
      if (v) el.setAttribute(k, "");
    } else el.setAttribute(k, v);
  }
  for (const c of children) if (c != null) el.append(typeof c === "string" ? document.createTextNode(c) : c);
  return el;
}
```

```ts
export type Page = (root: HTMLElement, params: string[]) => void | Promise<void>;
export type Route = { pattern: RegExp; page: Page };

let routes: Route[] = [];
let root: HTMLElement;
let notFound: Page;

export function navigate(path: string): void {
  history.pushState(null, "", path);
  render();
}

function render(): void {
  const path = location.pathname;
  for (const r of routes) {
    const m = path.match(r.pattern);
    if (m) {
      void r.page(root, m.slice(1));
      return;
    }
  }
  void notFound(root, []);
}

export function startRouter(el: HTMLElement, rs: Route[], nf: Page): void {
  root = el;
  routes = rs;
  notFound = nf;
  window.addEventListener("popstate", render);
  document.addEventListener("click", (ev) => {
    const a = (ev.target as HTMLElement).closest("a[data-link]") as HTMLAnchorElement | null;
    if (a && a.origin === location.origin) {
      ev.preventDefault();
      navigate(a.pathname + a.search);
    }
  });
  render();
}
```

```ts
import { ApiClient, ApiError, ChallengeIssued } from "./api";
import { h, clear, errorMessage } from "./dom";
import { t } from "./i18n";

const BOX = 66; // must match server Params.Body + 2*Knob

export class JigsawWidget {
  private root = h("div", { class: "jigsaw" });
  private attempts = 0;
  private challengeId = "";

  constructor(private api: ApiClient, private onToken: (token: string) => void) {}

  async mount(parent: HTMLElement): Promise<void> {
    parent.append(this.root);
    await this.load();
  }

  destroy(): void {
    this.root.remove();
  }

  private async load(): Promise<void> {
    clear(this.root);
    this.root.append(h("p", { class: "muted" }, t("challenge.loading")));
    let iss: ChallengeIssued;
    try {
      iss = await this.api.issueChallenge();
    } catch (e) {
      clear(this.root);
      this.root.append(h("p", { class: "error" }, errorMessage(e)));
      return;
    }
    this.challengeId = iss.challenge_id;
    clear(this.root);

    const stage = h("div", { class: "jigsaw-stage" });
    stage.style.width = `${iss.width}px`;
    stage.style.height = `${iss.height}px`;
    const bg = h("img", { class: "jigsaw-bg", alt: "", src: `data:image/png;base64,${iss.background_png_b64}`, draggable: "false" });
    const piece = h("img", { class: "jigsaw-piece", alt: "", src: `data:image/png;base64,${iss.piece_png_b64}`, draggable: "false" });
    piece.style.top = `${iss.piece_y}px`;
    piece.style.left = "0px";
    stage.append(bg, piece);

    const slider = h("input", { type: "range", min: "0", max: String(iss.width - BOX), value: "0", class: "jigsaw-slider", "aria-label": t("challenge.title") });
    slider.addEventListener("input", () => {
      piece.style.left = `${slider.value}px`;
    });
    slider.addEventListener("change", () => void this.submit(Number(slider.value)));

    this.root.append(h("p", {}, t("challenge.title")), stage, slider);
  }

  private async submit(x: number): Promise<void> {
    try {
      const res = await this.api.verifyChallenge(this.challengeId, x);
      this.onToken(res.challenge_token);
      clear(this.root);
      this.root.append(h("p", { class: "ok" }, "✓"));
    } catch (e) {
      this.attempts++;
      if (e instanceof ApiError && (e.status === 400 || e.status === 404) && this.attempts < 5) {
        this.root.prepend(h("p", { class: "error" }, t("challenge.retry")));
        await this.load();
        return;
      }
      clear(this.root);
      this.root.append(h("p", { class: "error" }, errorMessage(e)));
    }
  }
}
```

```ts
    if (s > 0) setTimeout(tick, 1000);
```

```ts
    pre, countdown, actions,
    h("p", { class: "muted small" }, t("view.copyNote")),

  );
```

```ts
import type { AppContext } from "../main";
import { canonicalBytes } from "../bytes";
import { h, clear, errorMessage } from "../dom";
import { sha256Hex } from "../hash";
import { t } from "../i18n";

const HEX64 = /^[0-9a-fA-F]{64}$/;

export function extractId(input: string): string | null {
  const m = input.trim().match(/([0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12})/);
  return m ? m[1]!.toLowerCase() : null;
}

export function verifyPage(ctx: AppContext) {
  return (root: HTMLElement) => {
    clear(root);
    const preset = new URLSearchParams(location.search).get("id") ?? "";
    const idInput = h("input", { placeholder: t("verify.id"), value: preset, "aria-label": t("verify.id") });
    const text = h("textarea", { rows: "8", placeholder: t("verify.text"), spellcheck: "false" });
    const file = h("input", { type: "file", accept: ".txt,text/plain" });
    const hash = h("input", { placeholder: t("verify.hash"), "aria-label": t("verify.hash") });
    const out = h("p", { class: "result", hidden: true });
    let fileBytes: Uint8Array | null = null;
    file.addEventListener("change", async () => {
      const f = file.files?.[0];
      fileBytes = f ? new Uint8Array(await f.arrayBuffer()) : null;
    });
    const form = h("form", {
      onSubmit: async (ev: Event) => {
        ev.preventDefault();
        out.hidden = true;
        const id = extractId(idInput.value);
        if (!id) {
          out.textContent = t("view.notFound");
          out.hidden = false;
          return;
        }
        try {
          let digest: string;
          if (HEX64.test(hash.value.trim())) digest = hash.value.trim().toLowerCase();
          else if (fileBytes) digest = await sha256Hex(withBOMBytes(fileBytes));
          else digest = await sha256Hex(canonicalBytes(text.value));
          const res = await ctx.api.verifyPaste(id, digest);
          out.textContent = res.match ? t("verify.match", { status: res.status }) : t("verify.noMatch");
          out.className = res.match ? "result ok" : "result error";
        } catch (e) {
          out.textContent = errorMessage(e);
          out.className = "result error";
        }
        out.hidden = false;
      },
    },
      h("h1", {}, t("verify.title")), idInput, text, h("label", {}, t("verify.file"), file), hash,
      h("button", { type: "submit" }, t("verify.submit")), out,
    );
    root.append(form);
  };
}

// Files are canonicalised exactly like text: prepend the BOM only if absent.
export function withBOMBytes(b: Uint8Array): Uint8Array {
  if (b.length >= 3 && b[0] === 0xef && b[1] === 0xbb && b[2] === 0xbf) return b;
  const out = new Uint8Array(b.length + 3);
  out.set([0xef, 0xbb, 0xbf], 0);
  out.set(b, 3);
  return out;
}
```
